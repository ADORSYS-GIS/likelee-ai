#### 3.3 Add useEffect to Load Billing Data
```typescript
useEffect(() => {
  if (activeSection !== "billing") return;
  let mounted = true;
  
  const loadBillingData = async () => {
    setLoadingBillingData(true);
    try {
      const [status, spend, invoices] = await Promise.all([
        getBrandBillingStatus(),
        getBrandSpendAnalytics(),
        listBrandInvoices()
      ]);
      if (mounted) {
        setBrandBillingStatus(status);
        setBrandSpendData(spend?.monthly_spend || []);
        setBrandInvoices(invoices?.invoices || []);
      }
    } catch (e) {
      // Handle error
    } finally {
      if (mounted) setLoadingBillingData(false);
    }
  };
  
  loadBillingData();
  return () => { mounted = false; };
}, [activeSection]);
```
